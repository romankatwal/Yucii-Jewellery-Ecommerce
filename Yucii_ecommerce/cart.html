Your cart is empty 🛒

Continue Shopping
`; return; } /* ========================================= GET PRODUCTS FROM DATABASE ========================================= */ fetch("/api/products") .then(response => { if (!response.ok) { throw new Error( "Failed to load products" ); } return response.json(); }) .then(products => { let subtotal = 0; let itemsHTML = ""; /* ========================================= CREATE EACH CART ITEM ========================================= */ cart.forEach((cartItem, index) => { const product = products.find( p => String(p.id) === String(cartItem.productId) ); // Product no longer exists if (!product) { return; } const quantity = Number(cartItem.quantity) || 1; const price = Number(product.price) || 0; const itemTotal = price * quantity; subtotal += itemTotal; /* ========================================= IMAGE ========================================= */ const imageName = product.image ? product.image .split("/") .pop() : ""; /* ========================================= CART ITEM HTML ========================================= */ itemsHTML += `

${quantity}
`; }); /* ========================================= TOTAL ========================================= */ const total = subtotal; /* ========================================= DISPLAY CART ========================================= */ cartContent.innerHTML = `
${itemsHTML}

`; }) .catch(error => { console.error( "Cart error:", error ); cartContent.innerHTML = `
